--TASK 1
SELECT 
    e.FIRST_NAME, 
    e.LAST_NAME, 
    e.SALARY, 
    m.FIRST_NAME AS MENECER_ADI, 
    m.SALARY AS MENECER_MAASI
FROM 
    HR.EMPLOYEES e
LEFT JOIN 
    HR.EMPLOYEES m ON e.MANAGER_ID = m.EMPLOYEE_ID;
--TASK 2
SELECT 
    FIRST_NAME, 
    LAST_NAME, 
    JOB_ID
FROM 
    HR.EMPLOYEES
WHERE 
    JOB_ID IN (
        SELECT JOB_ID 
        FROM HR.JOBS 
        WHERE JOB_TITLE IN ('Programmer', 'Accountant')
    );
--TASK 3
SELECT 
    E.FIRST_NAME,
    E.LAST_NAME
FROM 
    HR.EMPLOYEES E
JOIN 
    HR.EMPLOYEES M ON E.MANAGER_ID = M.EMPLOYEE_ID
WHERE 
    M.FIRST_NAME = 'Neena';
--TASK 4
SELECT 
    m.FIRST_NAME AS MENECER_ADI,
    m.LAST_NAME AS MENECER_SOYADI,
    COUNT(e.EMPLOYEE_ID) AS TABE_SAYI
FROM 
    HR.EMPLOYEES e
JOIN 
    HR.EMPLOYEES m ON e.MANAGER_ID = m.EMPLOYEE_ID
GROUP BY 
    m.EMPLOYEE_ID, 
    m.FIRST_NAME, 
    m.LAST_NAME
ORDER BY 
    TABE_SAYI DESC;
--TASK 5
SELECT
    EMPLOYEE_ID,
    FIRST_NAME,
    LAST_NAME
FROM
    HR.EMPLOYEES
WHERE EMPLOYEE_ID NOT IN (SELECT MANAGER_ID FROM HR.EMPLOYEES WHERE MANAGER_ID IS NOT NULL) --MANAGER ID SI NULLARI NOT IN QAYTARA BILMIR
--TASK 6
SELECT 
    FIRST_NAME, 
    DEPARTMENT_ID, 
    SALARY
FROM 
    HR.EMPLOYEES e
WHERE 
    SALARY < (
        SELECT AVG(SALARY)
        FROM HR.EMPLOYEES M
        WHERE M.DEPARTMENT_ID =E.DEPARTMENT_ID
    );
--TASK 7
SELECT
    DEPARTMENT_ID,
    MAX(SALARY) AS MAX_MAAS
FROM
    HR.EMPLOYEES
GROUP BY DEPARTMENT_ID;

SELECT
    DISTINCT E.DEPARTMENT_ID,
    (SELECT MAX(SALARY)
    FROM HR.EMPLOYEES
    WHERE DEPARTMENT_ID= E.DEPARTMENT_ID)
    AS MAX_MAAS
FROM HR.EMPLOYEES E;
--TASK 8
SELECT a.FIRST_NAME, a.SALARY
FROM HR.EMPLOYEES a
WHERE a.SALARY > (SELECT AVG(SALARY), DEPARTMENT_ID--DEPARTMENT_ID YE GORE 2 QIYMET QAYTARIR-OLMAZ
                  FROM EMPLOYEES a--HR.EMPLOYEES OLMALIDIR
                  WHERE a.DEPARTMENT_ID = a.DEPARTMENT_ID);--BIRI B OLMALIDIR OZU OZUNNEN HEMISE TRUE VERIR
--TASK 9
SELECT 
    e.FIRST_NAME,
    e.DEPARTMENT_ID,
    e.SALARY,
    ROUND(s.ORTA_MAAS) AS SOBE_ORTASI,
    (e.SALARY - ROUND(s.ORTA_MAAS)) AS FERQ
FROM 
    HR.EMPLOYEES e
JOIN (
    SELECT 
        DEPARTMENT_ID,
        AVG(SALARY) AS ORTA_MAAS
    FROM 
        HR.EMPLOYEES
    GROUP BY 
        DEPARTMENT_ID
) s ON e.DEPARTMENT_ID = s.DEPARTMENT_ID
WHERE 
    e.SALARY > s.ORTA_MAAS
ORDER BY 
    FERQ DESC;
