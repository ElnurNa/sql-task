--TASK 1
SELECT FIRST_NAME || ' ' || UPPER(LAST_NAME) AS TAM_AD,
       SUBSTR(FIRST_NAME, 1, 1) || '.' || ' ' || LAST_NAME AS QISA_AD,
       LOWER(EMAIL) || '@company.com'
FROM HR.EMPLOYEES
ORDER BY LAST_NAME;
--TASK 2
SELECT FIRST_NAME,
       TO_CHAR(SALARY,'fm99,999') || ' ' || 'AZN' AS MAAS,
       TO_CHAR(SALARY*12,'fm999,999')|| ' ' || 'AZN' AS ILLIK_MAAS
FROM
    HR.EMPLOYEES
ORDER BY MAAS DESC;--ÇIXAN SƏHV SƏTRDƏ RƏQƏMLƏRİN MÜQAİSƏSİ DEYİL, İLK SİMVOLLARIN MÜQAİSƏSİ OLUR,
--BU SƏBƏBDƏN 9 10 DAN BÖYÜK GÖTÜRÜLÜR 9>1
--TASK 3
SELECT TO_CHAR(HIRE_DATE,'DD,MM,YYYY') AS ISE_GIRIS,
       EXTRACT(YEAR FROM HIRE_DATE)AS IL,
       TO_CHAR(HIRE_DATE,'MONTH')AS AY_ADI,
       TO_CHAR(HIRE_DATE,'DAY')AS HEFTE_GUNU
FROM HR.EMPLOYEES
ORDER BY HIRE_DATE ASC;
--TASK 4
SELECT FIRST_NAME || LAST_NAME AS ISCI,
       TO_CHAR(HIRE_DATE,'DD') || ' ' || TO_CHAR(HIRE_DATE,'MONTH') || ' ' || TO_CHAR(HIRE_DATE,'YYYY') AS ISE_GIRIS,
       TRUNC(MONTHS_BETWEEN(SYSDATE, HIRE_DATE)/12) || ' ' || 'IL' || ' ' || MOD(TRUNC(MONTHS_BETWEEN(SYSDATE, HIRE_DATE)),12) || ' ' || 'AY' AS STAJ
FROM HR.EMPLOYEES
ORDER BY TRUNC(MONTHS_BETWEEN(SYSDATE, HIRE_DATE)) DESC;
--TASK 5
SELECT FIRST_NAME,HIRE_DATE,
       ADD_MONTHS(HIRE_DATE,3) AS SINAQ_BITIR,
       LAST_DAY(HIRE_DATE) AS ILK_MAAS,
       LAST_DAY(HIRE_DATE)-HIRE_DATE AS MAASA_QALAN_GUN
FROM HR.EMPLOYEES
SELECT ADD_MONTHS(TO_DATE('28.02.2025','DD.MM.YYYY'),3)
FROM DUAL;--28 FEVRALIN SON GUNUDUR DEYE SON GUN KIMI YADDA SAXLAYIR BUDA MAY AYININ SON GUNUNU GOTURMESINE SEBEB OLUR
--TASK 6
SELECT CONCAT(FIRST_NAME, ' ', LAST_NAME) || ', komissiya: ' || NVL(COMMISSION_PCT, 'yoxdur') AS MELUMAT--KOMISIYA OLAN SUTUNA SETIR DAXIL EDIR-XETA VERIR,fm0.00 olmasa 0.2 kimi yox .2 kimi gorunecek
FROM HR.EMPLOYEES
WHERE TO_CHAR(HIRE_DATE, 'MM') = '3';--03 verilmelidir cunki 2 olculudur
--CORRECTED
SELECT CONCAT(FIRST_NAME, ' ', LAST_NAME) || ', komissiya: ' || NVL(TO_CHAR(COMMISSION_PCT, 'fm0.00'), 'yoxdur') AS MELUMAT
FROM HR.EMPLOYEES
WHERE TO_CHAR(HIRE_DATE, 'MM') = '03';
--TASK 7
SELECT 
    TO_CHAR(HIRE_DATE,'MM') AS AY_NO,
    TRIM(TO_CHAR(HIRE_DATE,'MONTH')) AS AY,
    COUNT(*) AS ISCI_SAYI
FROM HR.EMPLOYEES
GROUP BY TO_CHAR(HIRE_DATE, 'MM'),TRIM(TO_CHAR(HIRE_DATE, 'MONTH'))
ORDER BY AY;--ELIFBA SIRASINA GORE, CUNKI SETRLERE GORE SIRALAYIR, SETRLERIN SIRASI DA NORMAL OLARAQ ELIFBA SIRASI ILE OLUR.
--TASK 8
SELECT
    LOWER(SUBSTR(FIRST_NAME,1,1) || LAST_NAME || LPAD(EMPLOYEE_ID,4,'0')) AS LOGIN,
    REPLACE(PHONE_NUMBER,'.','-') AS TELEFON
FROM HR.EMPLOYEES
WHERE LENGTH(LAST_NAME)>=7
ORDER BY LOGIN
--TASK 9
SELECT
    TO_CHAR(HIRE_DATE,'YYYY') AS IL,
    COUNT(*) AS ISCI_SAYI,
    TO_CHAR(MIN(HIRE_DATE), 'DD.MM.YYYY')AS ILK_TARIX,
    TO_CHAR(MAX(HIRE_DATE), 'DD.MM.YYYY') AS SON_TARIX,
    TO_CHAR(AVG(SALARY), 'fm999,999.00') AS ORTA_MAAS
FROM HR.EMPLOYEES
GROUP BY TO_CHAR(HIRE_DATE,'YYYY')
ORDER BY IL ASC;

SELECT
    FIRST_NAME,
    SUBSTR(TO_CHAR(HIRE_DATE,'DAY'),1,3)|| ',' || TO_CHAR(HIRE_DATE,'DD.MM.YYYY') AS ISE_GIRIS,
    CASE
        WHEN SUBSTR(TO_CHAR(HIRE_DATE,'DAY'),1,3)='SAT' OR SUBSTR(TO_CHAR(HIRE_DATE,'DAY'),1,3)='SUN' THEN 'HEFTE SONU'--ELE YAZSAQ BIZ BOSLUQLARI DA NEZERE ALMALI OLACAGIQ, BU SEBEBDEN SUBSTR YAXSIDIR
        ELSE 'IS GUNU'
    END GUN_NOVU
FROM HR.EMPLOYEES
--B
SELECT TRIM(TO_CHAR(HIRE_DATE, 'Day')) AS GUN,
       COUNT(*) AS ISCI_SAYI
FROM HR.EMPLOYEES
GROUP BY TRIM(TO_CHAR(HIRE_DATE, 'Day')),
         TO_CHAR(HIRE_DATE, 'D')
ORDER BY ISCI_SAYI DESC;
