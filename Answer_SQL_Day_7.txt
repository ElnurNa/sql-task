SELECT first_name, last_name, department_id
FROM HR.EMPLOYEES
WHERE department_id IN (
    SELECT department_id
    FROM HR.DEPARTMENTS
    WHERE department_name IN ('Marketing', 'Purchasing', 'IT')
)
SELECT E.FIRST_NAME , E.LAST_NAME , E.DEPARTMENT_ID
FROM HR.EMPLOYEES E
WHERE EXISTS(
    SELECT 1 FROM HR.DEPARTMENTS D
    WHERE D.department_id = E.department_id AND D.department_name IN ('Marketing', 'Purchasing', 'IT')
)
