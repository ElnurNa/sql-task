-- TASK 1
SELECT JOB_ID, 
    COUNT(*) AS ISCI_SAYI,
    FLOOR(AVG(SALARY)) AS ORTA_MAAS 
FROM HR.EMPLOYEES
GROUP BY JOB_ID
ORDER BY ISCI_SAYI DESC;

-- TASK 2
SELECT DEPARTMENT_ID,
       MIN(SALARY) AS MIN_MAAS,
       MAX(SALARY) AS MAX_MAAS
FROM HR.EMPLOYEES
GROUP BY DEPARTMENT_ID
ORDER BY DEPARTMENT_ID ASC;

-- TASK 3
SELECT D.DEPARTMENT_NAME, L.CITY,D.LOCATION_ID,L.LOCATION_ID
FROM HR.DEPARTMENTS D
JOIN HR.LOCATIONS L
ON D.LOCATION_ID=L.LOCATION_ID;

-- TASK 4
SELECT SUM(SALARY) AS MAAS_CEMI,
       JOB_ID
FROM HR.EMPLOYEES
GROUP BY JOB_ID
ORDER BY MAAS_CEMI DESC;

-- TASK 5
SELECT MANAGER_ID,
       COUNT(EMPLOYEE_ID) AS ISCI_SAYI
FROM HR.EMPLOYEES
WHERE COMMISSION_PCT IS NOT NULL
GROUP BY MANAGER_ID
HAVING COUNT(EMPLOYEE_ID) > 5;
--SELECT * FROM HR.EMPLOYEES;

-- TASK 6
SELECT L.LOCATION_ID,
       D.LOCATION_ID,
       L.CITY,
       COUNT(D.DEPARTMENT_ID) AS SOBE_SAYI
FROM HR.LOCATIONS L
LEFT JOIN HR.DEPARTMENTS D
ON L.LOCATION_ID = D.LOCATION_ID
GROUP BY L.LOCATION_ID,
         D.LOCATION_ID,
         L.CITY
ORDER BY SOBE_SAYI DESC;
-- TASK 7
SELECT E.FIRST_NAME,
       E.LAST_NAME,
       D.DEPARTMENT_NAME
FROM HR.EMPLOYEES E
FULL JOIN HR.DEPARTMENTS D
ON E.DEPARTMENT_ID=D.DEPARTMENT_ID
WHERE E.DEPARTMENT_ID IS NULL OR D.DEPARTMENT_ID IS NULL;

-- TASK 8
--WRONG VERSION

FROM HR.EMPLOYEES E

--CORRECTED
SELECT D.DEPARTMENT_NAME, COUNT(*) AS ISCI_SAYI
FROM HR.EMPLOYEES E
JOIN HR.DEPARTMENTS D ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
GROUP BY D.DEPARTMENT_NAME
HAVING COUNT(*) > 3;
-- TASK 9
SELECT D.DEPARTMENT_NAME,
       J.JOB_TITLE,
       COUNT(*) AS ISCI_SAYI,
       CEIL(AVG(E.SALARY)) AS ORTA_MAAS
FROM HR.EMPLOYEES E
JOIN HR.DEPARTMENTS D ON E.DEPARTMENT_ID=D.DEPARTMENT_ID
JOIN HR.JOBS J ON E.JOB_ID =J.JOB_ID
GROUP BY D.DEPARTMENT_NAME,J.JOB_TITLE
HAVING COUNT(*)>=3
ORDER BY D.DEPARTMENT_NAME ASC, ISCI_SAYI DESC;

-- TASK 10
SELECT E.EMPLOYEE_ID, 
       E.FIRST_NAME, 
       E.LAST_NAME, 
       COUNT(H.EMPLOYEE_ID) AS KECMIS_VEZIFE_SAYI
FROM HR.EMPLOYEES E
LEFT JOIN HR.JOB_HISTORY H 
  ON E.EMPLOYEE_ID = H.EMPLOYEE_ID
GROUP BY E.EMPLOYEE_ID, E.FIRST_NAME, E.LAST_NAME
ORDER BY KECMIS_VEZIFE_SAYI DESC;

--CITY,DEPARTMENT ID,AVG SALARY,LOCATION_ID
SELECT
       L.CITY,
       CEIL(AVG(E.SALARY)) AS ORTA_MAAS
FROM HR.EMPLOYEES E
JOIN HR.DEPARTMENTS D
ON E.DEPARTMENT_ID = D.DEPARTMENT_ID
JOIN HR.LOCATIONS L
ON D.LOCATION_ID=L.LOCATION_ID
GROUP BY L.CITY
