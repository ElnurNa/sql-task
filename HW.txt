--TASK 1
SELECT FIRST_NAME,
       SALARY,
       CASE
        WHEN SALARY<3000 THEN 'COX ASAGI'
        WHEN SALARY BETWEEN 3000 AND 6999 THEN 'ASAGI'
        WHEN SALARY BETWEEN 7000 AND 11999 THEN 'ORTA'
        ELSE 'YUKSEK'
       END AS KATEQORIYA
FROM HR.EMPLOYEES
--TASK 2
SELECT 
    FIRST_NAME, 
    DEPARTMENT_ID,
    CASE DEPARTMENT_ID
        WHEN 10 THEN 'Admin'
        WHEN 20 THEN 'Marketing'
        WHEN 60 THEN 'IT'
        WHEN 90 THEN 'Rehberlik'
        ELSE 'Diger'
    END AS SOBE_ADI
FROM 
    HR.EMPLOYEES;
--TASK 3
SELECT
    FIRST_NAME,
    NVL(MANAGER_ID,0) AS MANAGER_ID,
    NVL(COMMISSION_PCT,0) AS KOMISSIYA
FROM HR.EMPLOYEES;
--TASK 4
SELECT
    JOB_TITLE,
    MIN_SALARY,
    MAX_SALARY,
    CASE
        WHEN (MAX_SALARY-MIN_SALARY)<5000 THEN 'DAR'
        WHEN (MAX_SALARY-MIN_SALARY) BETWEEN 5000 AND 10000 THEN 'ORTA'
        ELSE 'GENIS'
    END ARALİQ
FROM HR.JOBS
--TASK 5
SELECT
     FIRST_NAME,
     SALARY,
     SALARY*(1+NVL(COMMISSION_PCT,0)) AS AYLIQ_GELIR,
     CASE 
        WHEN COMMISSION_PCT IS NULL THEN 'DIGER'
        ELSE 'SATICI'
     END NOV
FROM HR.EMPLOYEES
--TASK 6
--DUZELDİLMİS VERSİYA
SELECT FIRST_NAME, SALARY,
       CASE
           WHEN COMMISSION_PCT IS NULL THEN 'Komissiyasiz'
           ELSE 'KOMISSIYALI'
       END AS "Komissiya"
FROM HR.EMPLOYEES;
--İLKİN VERSİYA
SELECT FIRST_NAME, SALARY,
       CASE COMMISSION_PCT--COMMİSİON WHENLERIN ICINDE VERILMELI IDI
           WHEN NULL THEN 'Komissiyasiz'--COMMISION YAZILDIQDAN SONRA IS NULL OLACAQ
           ELSE COMMISSION_PCT--KOMISSIYALI QAYTARMALIDIR
       END AS 'Komissiya'--CUT DIRNAQLA YAZILIR
FROM HR.EMPLOYEES;
--TASK 7
SELECT
    FIRST_NAME,
    JOB_ID,
    SALARY,
    CASE
        WHEN JOB_ID LIKE 'AD%' THEN 'REHBERLIK'
        WHEN (JOB_ID LIKE '%MAN' OR JOB_ID LIKE '%MGR') AND SALARY>12000 THEN 'BOYUK MENECER'
        WHEN JOB_ID LIKE '%MAN' OR JOB_ID LIKE '%MGR' THEN 'MENECER'
        ELSE 'ISCI'
        END "ROL"
FROM HR.EMPLOYEES
--TASK 8
SELECT 
    FIRST_NAME, 
    SALARY, 
    COMMISSION_PCT,
    CASE 
        WHEN NVL(COMMISSION_PCT, 0) >= 0.3 THEN 'Yuksek'
        WHEN NVL(COMMISSION_PCT, 0) >= 0.2 THEN 'Orta'
        ELSE 'Asagi'
    END AS KOMISSIYA_SEVIYYESI
FROM 
    HR.EMPLOYEES
WHERE 
    DEPARTMENT_ID = 80
ORDER BY 
    KOMISSIYA_SEVIYYESI DESC, 
    SALARY DESC;
